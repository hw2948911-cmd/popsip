// PopSip 排行榜 Worker —— 部署到 Cloudflare Workers（免费版够用）
// 环境变量（Secrets）：
//   ZALO_BOT_TOKEN   机器人 token（先不做自动发榜可以先不填）
//   ZALO_CHAT_ID     群 ID（同上）
//   ADMIN_KEY        后台口令，用来把“已领奖”标记打上
// KV 绑定：SCORES（命名空间变量名必须叫 SCORES）

const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
  'Content-Type': 'application/json; charset=utf-8'
};
const json = (o, s = 200) => new Response(JSON.stringify(o), { status: s, headers: CORS });

// 越南时区当天 0 点（UTC+7）
function dayKey(ts = Date.now()) {
  const d = new Date(ts + 7 * 3600 * 1000);
  return d.toISOString().slice(0, 10);           // YYYY-MM-DD
}
function weekKey(ts = Date.now()) {
  const d = new Date(ts + 7 * 3600 * 1000);
  const date = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()));
  const dow = (date.getUTCDay() + 6) % 7;        // 周一 = 0
  date.setUTCDate(date.getUTCDate() - dow);
  return date.toISOString().slice(0, 10);
}

// 防刷：单局理论最高分。超出直接丢弃
const MAX_SCORE = 20000;
const MAX_COMBO = 400;

async function submitScore(env, body) {
  const name = String(body.name || '').slice(0, 16).trim();
  const score = Math.floor(Number(body.score));
  const combo = Math.min(Math.floor(Number(body.combo) || 0), MAX_COMBO);
  const hidden = body.hidden ? 1 : 0;
  const dev = String(body.dev || '').slice(0, 24);
  if (!name || !Number.isFinite(score) || score < 0 || score > MAX_SCORE) {
    return json({ ok: false, error: 'bad_payload' }, 400);
  }
  const dk = dayKey(), wk = weekKey();

  // 一人一天只保留最高分
  const dayEntryKey = `d:${dk}:${name}`;
  const prev = Number(await env.SCORES.get(dayEntryKey)) || -1;
  if (score <= prev) return json({ ok: true, kept: false, best: prev });

  await env.SCORES.put(dayEntryKey, String(score));
  await env.SCORES.put(`m:${dk}:${name}`, JSON.stringify({ name, score, combo, hidden, ts: Date.now() }));

  // 当日榜（前 100 名，按分数→连击排序）
  const listRaw = (await env.SCORES.get(`lb:${dk}`)) || '[]';
  const list = JSON.parse(listRaw).filter(e => e.name !== name);
  list.push({ name, score, combo, hidden, ts: Date.now() });
  list.sort((a, b) => b.score - a.score || b.combo - a.combo);
  await env.SCORES.put(`lb:${dk}`, JSON.stringify(list.slice(0, 100)), { expirationTtl: 60 * 60 * 24 * 10 });

  // 周榜
  const wRaw = (await env.SCORES.get(`wl:${wk}`)) || '[]';
  const wl = JSON.parse(wRaw).filter(e => e.name !== name);
  wl.push({ name, score, combo, ts: Date.now() });
  wl.sort((a, b) => b.score - a.score || b.combo - a.combo);
  await env.SCORES.put(`wl:${wk}`, JSON.stringify(wl.slice(0, 100)), { expirationTtl: 60 * 60 * 24 * 40 });

  const rank = list.findIndex(e => e.name === name) + 1;
  return json({ ok: true, kept: true, rank, total: list.length });
}

async function getBoard(env, url) {
  const dk = url.searchParams.get('day') || dayKey();
  const name = (url.searchParams.get('name') || '').slice(0, 16);
  const list = JSON.parse((await env.SCORES.get(`lb:${dk}`)) || '[]');
  const top = list.slice(0, 10).map((e, i) => ({ rank: i + 1, name: e.name, score: e.score, combo: e.combo }));
  let me = null;
  if (name) {
    const i = list.findIndex(e => e.name === name);
    if (i >= 0) me = { rank: i + 1, name, score: list[i].score, combo: list[i].combo, total: list.length };
  }
  return json({ ok: true, day: dk, top, me });
}

async function zaloSend(env, text) {
  if (!env.ZALO_BOT_TOKEN || !env.ZALO_CHAT_ID) return { skipped: true };
  const r = await fetch(`https://bot-api.zapps.me/bot${env.ZALO_BOT_TOKEN}/sendMessage`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ chat_id: env.ZALO_CHAT_ID, text })
  });
  return { status: r.status, body: (await r.text()).slice(0, 200) };
}

function detectLang(text){
  if (/[\u4e00-\u9fff]/.test(text)) return 'zh';
  if (/[àáảãạăâđèéẻẽẹêìíỉĩịòóỏõọôơùúủũụưỳýỷỹỵ]/i.test(text)) return 'vi';
  return 'vi';
}

function pickReply(text) {
  const lang = detectLang(text);
  for (const q of QUICK) {
    if (q.re.test(text)) return lang === 'zh' ? q.zh : q.vi;
  }
  return null;
}

// 记录用量（估算），防止超额
async function bumpUsage(env, tokens) {
  const k = 'usage:' + new Date().toISOString().slice(0, 7); // YYYY-MM
  const cur = Number(await env.SCORES.get(k)) || 0;
  await env.SCORES.put(k, String(cur + tokens), { expirationTtl: 60 * 60 * 24 * 70 });
  return cur + tokens;
}

async function aiReply(env, convKey, userText, displayName) {
  if (!env.DEEPSEEK_API_KEY) return null;
  const hist = JSON.parse((await env.SCORES.get(convKey)) || '[]');
  const messages = [{ role: 'system', content: PERSONA }];
  for (const h of hist.slice(-6)) messages.push(h);
  const lang = detectLang(userText);
  const force = lang === 'zh'
    ? 'BẮT BUỘC: trả lời bằng TIẾNG TRUNG (简体中文), không dùng tiếng Việt.'
    : 'BẮT BUỘC: trả lời bằng TIẾNG VIỆT.';
  messages.push({ role: 'system', content: force });
  messages.push({ role: 'user', content: `${displayName}: ${userText}` });

  const r = await fetch('https://api.deepseek.com/chat/completions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + env.DEEPSEEK_API_KEY },
    body: JSON.stringify({ model: 'deepseek-chat', messages, max_tokens: 120, temperature: 1.1 })
  });
  const d = await r.json();
  const out = d && d.choices && d.choices[0] && d.choices[0].message && d.choices[0].message.content;
  if (!out) return null;

  hist.push({ role: 'user', content: `${displayName}: ${userText}` });
  hist.push({ role: 'assistant', content: out });
  await env.SCORES.put(convKey, JSON.stringify(hist.slice(-10)), { expirationTtl: 60 * 60 * 24 * 3 });
  if (d.usage) await bumpUsage(env, (d.usage.total_tokens || 0));
  return out.trim();
}

// 群内限频：每组每分钟最多 6 条
async function rateOk(env, chatId) {
  const minute = new Date().toISOString().slice(0, 16);
  const k = `rl:${chatId}:${minute}`;
  const n = Number(await env.SCORES.get(k)) || 0;
  if (n >= 6) return false;
  await env.SCORES.put(k, String(n + 1), { expirationTtl: 180 });
  return true;
}

async function handleMessage(env, up, chatId) {
  try {
    const m = up.message || {};
    const from = m.from || {};
    if (from.is_bot) return;
    const raw = (m.text || '').trim();
    if (!raw) return;
    const isDM = m.chat && m.chat.chat_type === 'PRIVATE';
    // 群里必须点名才理
    if (!isDM && raw.indexOf('@') < 0) return;
    const text = raw.replace(/@\S+/g, '').trim();
    if (!text) return;
    if (!(await rateOk(env, chatId))) return;

    const name = from.display_name || 'bạn';
    const convKey = `conv:${chatId}:${from.id || name}`;
    let reply = null;
    if (/签到|điểm danh|check\s*in/i.test(text)) {
      from.__text = text;
      reply = await doCheckin(env, chatId, from);
    } else {
      reply = pickReply(text);
    }
    if (!reply) reply = await aiReply(env, convKey, text, name);
    if (!reply) reply = 'Ừm... mình chưa hiểu lắm, nói lại giúp mình nha 🙈';

    const delay = 900 + Math.random() * 1600;   // 真人打字节奏
    await new Promise(r => setTimeout(r, delay));
    await zaloSendTo(env, chatId, reply);
  } catch (e) {}
}


// ================= 签到 & 早安播报 =================
async function doCheckinBase(env, chatId, from) {
  const dk = dayKey();
  const uid = from.id || from.display_name || 'u';
  const who = from.display_name || 'bạn';
  const lang = detectLang(from.__text || '');

  // 一天一次
  const dayLock = `ckday:${uid}:${dk}`;
  if (await env.SCORES.get(dayLock)) {
    return lang === 'zh'
      ? [
          `${who} 贪心哦～今天已经签过啦 😤 明天再来找我`,
          `诶诶诶，同一天不能签两次的！我只认一次 😌`,
          `你今天已经出现过啦，我可记得很清楚 👀 明天见`
        ][Math.floor(Math.random() * 3)]
      : [
          `Này này ${who}, hôm nay ký rồi mà 😤 Mai quay lại nha`,
          `Ơ kìa, một ngày chỉ điểm danh một lần thôi nha 😌`,
          `Tôi nhớ mặt bạn rồi đó 👀 Mai gặp lại`
        ][Math.floor(Math.random() * 3)];
  }
  await env.SCORES.put(dayLock, '1', { expirationTtl: 60 * 60 * 36 });

  // 累计计数（不清零）
  const totalKey = `ci:total:${uid}`;
  const n = (Number(await env.SCORES.get(totalKey)) || 0) + 1;
  await env.SCORES.put(totalKey, String(n), { expirationTtl: 60 * 60 * 24 * 120 });

  // 满 5 次 → 兑换判定（每人每周最多 1 次）
  if (n >= 5) {
    const lastKey = `ci:claim:${uid}`;
    const last = Number(await env.SCORES.get(lastKey)) || 0;
    const weekMs = 7 * 24 * 3600 * 1000;
    if (last && Date.now() - last < weekMs) {
      return lang === 'zh'
        ? `满了 5 次！但 ${who} 这周已经领过蛋挞冰淇淋了 😤 我替你留着，下周再来`
        : `Đủ 5 lần rồi đó ${who}! Nhưng tuần này bạn đã nhận bánh tart kem rồi 😤 Tôi giữ sổ, tuần sau quay lại nha`;
    }
    await env.SCORES.put(lastKey, String(Date.now()), { expirationTtl: 60 * 60 * 24 * 400 });
    await env.SCORES.put(totalKey, '0', { expirationTtl: 60 * 60 * 24 * 120 });
    const t = Number(await env.SCORES.get(`ci:grand:${dk.slice(0, 7)}`)) || 0;
    await env.SCORES.put(`ci:grand:${dk.slice(0, 7)}`, String(t + 1), { expirationTtl: 60 * 60 * 24 * 70 });
    return lang === 'zh'
      ? `🎉 ${who} 集满 5 次啦！把这条给店员看，能拿一份蛋挞冰淇淋（25K）。下次重新开始攒哦`
      : `🎉 Đủ 5 lần rồi ${who}! Đưa tin này cho nhân viên để nhận 1 bánh tart kem (25K) nha. Lần sau tích tiếp từ đầu 😌`;
  }

  const left = 5 - n;
  if (left === 1) {
    return lang === 'zh'
      ? `再签 **1 次** 就能换蛋挞冰淇淋了哦～明天记得来 😏`
      : `Còn đúng **1 lần** nữa là có bánh tart kem đó nha, mai nhớ quay lại 😏`;
  }
  const vi = [
    `Ơ ${who} nay siêng thế! Lần ${n} rồi nha ✍️ Còn ${left} lần nữa là có bánh tart kem (25K)`,
    `Điểm danh lần ${n} cho ${who} ✅ Còn ${left} lần là có bánh tart kem (25K) rồi`,
    `${who} điểm danh rồi hả, nhớ dai ghê 😌 Lần ${n} nhé, còn ${left} lần là có bánh tart kem`,
    `Lần ${n} nhận rồi nha ${who}! Tôi đếm từng lần đấy, đừng hòng gian lận 😤 Còn ${left} lần là có bánh tart kem`
  ];
  const zh = [
    `哦～${who} 今天很勤快嘛！第 ${n} 次记上了 ✍️ 还差 ${left} 次就能换蛋挞冰淇淋（25K）`,
    `${who} 签到成功 ✅ 第 ${n} 次，再 ${left} 次就有蛋挞冰淇淋`,
    `${who} 来啦，我可都记着的 😌 第 ${n} 次，还差 ${left} 次换蛋挞冰淇淋`,
    `第 ${n} 次！我一次一次数的，别想蒙我 😤 还差 ${left} 次就能换蛋挞冰淇淋`
  ];
  const arr = lang === 'zh' ? zh : vi;
  return arr[Math.floor(Math.random() * arr.length)];
}

function withChatter(mainReply, lang) {
  const r = Math.random();
  let extra = '';
  if (r < 0.25) {
    const promo = lang === 'zh' ? PROMO_ZH : PROMO_VI;
    if (promo) extra = promo;
  }
  if (!extra && r < 0.78) {
    const pool = lang === 'zh' ? CHATTER_ZH : CHATTER_VI;
    extra = pool[Math.floor(Math.random() * pool.length)];
  }
  return extra ? mainReply + '\n\n' + extra : mainReply;
}

  const base = await doCheckinBase(env, chatId, from);

}

// ================= 胡志明市天气 =================
const WMO_VI = (c) => {
  if (c === 0) return 'trời quang';
  if (c <= 3) return 'có mây';
  if (c === 45 || c === 48) return 'có sương mù';
  if (c >= 51 && c <= 57) return 'mưa phùn';
  if (c >= 61 && c <= 67) return 'mưa';
  if (c >= 80 && c <= 82) return 'mưa rào';
  if (c >= 95) return 'mưa dông';
  return 'trời dịu';
};

async function getWeather(env) {
  const cached = await env.SCORES.get('weather:hcm');
  if (cached) { try { return JSON.parse(cached); } catch (_) {} }
  try {
    const u = 'https://api.open-meteo.com/v1/forecast?latitude=10.7626&longitude=106.6602'
      + '&current=temperature_2m,relative_humidity_2m,apparent_temperature,weather_code,precipitation'
      + '&daily=temperature_2m_max,temperature_2m_min,precipitation_probability_max,weather_code'
      + '&timezone=Asia%2FHo_Chi_Minh&forecast_days=1';
    const r = await fetch(u);
    const d = await r.json();
    const cur = d && d.current, day = d && d.daily;
    if (!cur || !day) return null;
    const w = {
      temp: Math.round(cur.temperature_2m),
      feels: Math.round(cur.apparent_temperature),
      hum: Math.round(cur.relative_humidity_2m),
      code: cur.weather_code,
      rainP: Math.round((day.precipitation_probability_max || [0])[0]),
      tmax: Math.round((day.temperature_2m_max || [0])[0]),
      tmin: Math.round((day.temperature_2m_min || [0])[0]),
      ts: Date.now()
    };
    await env.SCORES.put('weather:hcm', JSON.stringify(w), { expirationTtl: 900 });
    return w;
  } catch (e) { return null; }
}

function weatherLine(w) {
  if (!w) return '';
  const desc = WMO_VI(w.code);
  const rainy = w.rainP >= 55 || (w.code >= 61);
  let comfort;
  if (w.feels >= 34) comfort = 'nóng nực';
  else if (w.feels >= 30) comfort = 'hơi nóng';
  else if (w.feels >= 26) comfort = 'dễ chịu';
  else comfort = 'mát';

  let tip;
  if (w.code >= 95) tip = 'chiều có mưa dông, ai ra đường nhớ mang áo mưa nha';
  else if (rainy) tip = `chiều có thể mưa (${w.rainP}%), nhớ mang áo mưa`;
  else if (w.feels >= 34) tip = 'nắng gắt, uống nhiều nước và tránh nắng giữa trưa';
  else if (w.feels >= 30) tip = 'hơi oi, chọn đồ uống có đá là chuẩn bài';
  else if (w.feels >= 26) tip = 'thời tiết dễ chịu, hợp ngồi quán chơi một ván';
  else tip = 'trời mát, dễ chịu, đi đâu cũng tiện';

  return `🌤 TP.HCM: ${w.temp}°C (cảm giác ${w.feels}°C, ${desc}) — ${tip}.`;
}

function vnNow() { return new Date(Date.now() + 7 * 3600 * 1000); }

async function maybeCheckinReminder(env) {
  const now = vnNow();
  const day = now.toISOString().slice(0, 10);
  const slot = now.getUTCHours() * 2 + (now.getUTCMinutes() >= 30 ? 1 : 0); // 09:00起每30分钟
  const seqKey = `ci:seq:${day}`;
  let plan = JSON.parse((await env.SCORES.get(seqKey)) || 'null');
  if (!plan) {
    // 09:00-21:30 之间共 26 个半小时槽，随机挑 3 个
    const slots = [];
    while (slots.length < 3) {
      const x = 18 + Math.floor(Math.random() * 26); // 02:00UTC=18 起算
      if (!slots.includes(x)) slots.push(x);
    }
    slots.sort((a, b) => a - b);
    plan = slots;
    await env.SCORES.put(seqKey, JSON.stringify(plan), { expirationTtl: 60 * 60 * 30 });
  }
  if (!plan.includes(slot)) return;
  const sentKey = `ci:sent:${day}:${slot}`;
  if (await env.SCORES.get(sentKey)) return;
  await env.SCORES.put(sentKey, '1', { expirationTtl: 60 * 60 * 30 });
  const text = CI_POOL_VI[Math.floor(Math.random() * CI_POOL_VI.length)];
  await zaloSend(env, text);
}

export default {
  async fetch(req, env, ctx) {
    const url = new URL(req.url);
    if (req.method === 'OPTIONS') return new Response(null, { headers: CORS });

    if (url.pathname === '/submit' && req.method === 'POST') {
      let body = {};
      try { body = await req.json(); } catch (_) { return json({ ok: false, error: 'json' }, 400); }
      return submitScore(env, body);
    }
    if (url.pathname === '/board') return getBoard(env, url);

    // 后台：标记已领奖（配合顺延规则）
    if (url.pathname === '/claim' && req.method === 'POST') {
      let body = {};
      try { body = await req.json(); } catch (_) {}
      if (body.key !== env.ADMIN_KEY) return json({ ok: false, error: 'auth' }, 401);
      const wk = weekKey();
      let claimed = JSON.parse((await env.SCORES.get(`claimed:${wk}`)) || '[]');
      const nm = String(body.name || '').slice(0, 16);
      if (body.on === false) claimed = claimed.filter(n => n !== nm);
      else if (nm && !claimed.includes(nm)) claimed.push(nm);
      await env.SCORES.put(`claimed:${wk}`, JSON.stringify(claimed), { expirationTtl: 60 * 60 * 24 * 40 });
      return json({ ok: true, claimed });
    }
    if (url.pathname === '/claimed') {
      const wk = weekKey();
      return json({ ok: true, week: wk, claimed: JSON.parse((await env.SCORES.get(`claimed:${wk}`)) || '[]') });
    }

    if (url.pathname === '/zalo-webhook' && req.method === 'POST') {
      let up = {};
      try { up = await req.json(); } catch (_) {}
      // 尽量兼容几种 payload 结构
      const chat = up.chat || (up.message && up.message.chat) || (up.data && up.data.chat) || {};
      const chatId = chat.id || up.chat_id || (up.data && up.data.chat_id);
      if (chatId) {
        await env.SCORES.put('bot:lastchat', JSON.stringify({ id: String(chatId), ts: Date.now(), raw: up }));
        await env.SCORES.put('bot:chat:' + chatId, String(Date.now()), { expirationTtl: 60 * 60 * 24 * 60 });
        if (up.event_name === 'message.text.received' && ctx && ctx.waitUntil) {
          ctx.waitUntil(handleMessage(env, up, String(chatId)));
        }
      }
      return json({ ok: true });
    }

    // 查看最近一次收到的群信息（需要 ?k=后台口令）
    if (url.pathname === '/debug/chat') {
      if (url.searchParams.get('k') !== env.ADMIN_KEY) return json({ ok: false, error: 'auth' }, 401);
      return json({ ok: true, last: JSON.parse((await env.SCORES.get('bot:lastchat')) || 'null') });
    }
    // 后台看板：今日榜 + 本周已领名单
    if (url.pathname === '/winners') {
      if (url.searchParams.get('k') !== env.ADMIN_KEY) return json({ ok: false, error: 'auth' }, 401);
      const dk = dayKey(), wk = weekKey();
      const today = JSON.parse((await env.SCORES.get(`lb:${dk}`)) || '[]').slice(0, 10);
      const weekTop = JSON.parse((await env.SCORES.get(`wl:${wk}`)) || '[]').slice(0, 5);
      const claimed = JSON.parse((await env.SCORES.get(`claimed:${wk}`)) || '[]');
      return json({ ok: true, day: dk, week: wk, today, weekTop, claimed });
    }
    // 给玩家分配本周唯一编号
    if (url.pathname === '/assign') {
      const wk = weekKey();
      const k = `seq:${wk}`;
      let n = Number(await env.SCORES.get(k)) || 1000;
      n += 1;
      await env.SCORES.put(k, String(n), { expirationTtl: 60 * 60 * 24 * 45 });

    if (url.pathname === '/selftest') {
      if (url.searchParams.get('k') !== env.ADMIN_KEY) return json({ ok: false, error: 'auth' }, 401);
      const morning = await buildMorning(env);
